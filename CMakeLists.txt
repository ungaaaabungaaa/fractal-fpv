cmake_minimum_required(VERSION 3.20)
project(FractalPortable LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(fractal_core INTERFACE)
target_include_directories(fractal_core INTERFACE Source/FractalFPV/FractalCore)
target_compile_features(fractal_core INTERFACE cxx_std_17)
add_executable(fractal-export tools/exporter.cpp)
target_link_libraries(fractal-export PRIVATE fractal_core)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/Source/FractalFPV/FractalCore/FractalCore.h"
  "${CMAKE_CURRENT_SOURCE_DIR}/tools/exporter.cpp")
file(SHA256 Source/FractalFPV/FractalCore/FractalCore.h CORE_HASH)
file(SHA256 tools/exporter.cpp TOOL_HASH)
string(SHA256 PRODUCER_HASH "${CORE_HASH}${TOOL_HASH}")
target_compile_definitions(fractal-export PRIVATE FRACTAL_CORE_SHA256="${CORE_HASH}" FRACTAL_PRODUCER_SHA256="${PRODUCER_HASH}")
add_executable(core-tests tests/core_tests.cpp)
target_link_libraries(core-tests PRIVATE fractal_core)
foreach(target fractal-export core-tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME core-invariants COMMAND core-tests)
